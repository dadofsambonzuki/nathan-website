{{/* Turn the raw nostr:npub1… / nostr:nprofile1… / nostr:nevent1… mentions that come through the Nostr sync into links, labelled with the profile's name when scripts/resolve_nostr_mentions.py has cached one in data/nostr_profiles.json (site.Data.nostr_profiles). Profiles link to nosta.me, events and addresses to jumble.social — the same hosts the rest of the site uses. Mentions the resolver could not decode (they get pasted truncated) are left as plain text rather than linked to a dead page. Call with the rendered content: {{ partial "nostr-mentions.html" .Content }} Pass media=true (as a dict: {{ partial "nostr-mentions.html" (dict "content" .Content "media" true) }}) on the notes, where a bare image or video URL in the post should render inline rather than as a link. */}} {{- $content := "" -}} {{- $media := false -}} {{- if reflect.IsMap . -}} {{- $content = .content | string -}} {{- $media = .media | default false -}} {{- else -}} {{- $content = . | string -}} {{- end -}} {{- $profiles := site.Data.nostr_profiles | default (dict) -}} {{- $pattern := `(?:npub|nprofile|note|nevent|naddr)1[023456789acdefghjklmnpqrstuvwxyz]{20,}` -}} {{/* Notes often carry media as a bare URL, which Hugo turns into a link; show it instead. Matches the whole anchor so the href is not rewritten into a second copy. */}} {{- if $media -}} {{- $content = $content | replaceRE ` ]+\.(?:jpg|jpeg|png|gif|webp))"?[^>]*>[^<]*` `` -}} {{- $content = $content | replaceRE ` ]+\.(?:mp4|webm))"?[^>]*>[^<]*` `` -}} {{- end -}} {{/* Cloak the undecodable mentions so the link passes below skip them. */}} {{- $invalid := slice -}} {{- range $id, $entry := $profiles -}} {{- if $entry.invalid -}} {{- $invalid = $invalid | append $id -}} {{- end -}} {{- end -}} {{- range $index, $id := $invalid -}} {{- $cloak := printf "@@nostr%d@@" $index -}} {{- $content = replace $content (printf "nostr:%s" $id) $cloak -}} {{- $content = replace $content $id $cloak -}} {{- end -}} {{/* Each mention becomes one link: the profile name when the resolver found one, otherwise the canonical short form from the cache — a plain npub for a profile, a note1 for an event — so a nameless mention still reads as a short identifier rather than a 200-character nprofile or nevent. */}} {{- range $index, $id := (findRE $pattern $content | uniq) -}} {{- $entry := index $profiles $id -}} {{- $target := $id -}} {{- with $entry -}} {{- with .short -}} {{- $target = . -}} {{- end -}} {{- end -}} {{- $label := "" -}} {{- with $entry -}} {{- $label = or .display_name .name .nip05 "" -}} {{- end -}} {{- if not $label -}} {{- $label = printf "%s…%s" (substr $target 0 10) (substr $target (sub (len $target) 4)) -}} {{- end -}} {{- $host := "https://jumble.social/" -}} {{- if or (hasPrefix $target "npub") (hasPrefix $target "nprofile") -}} {{- $host = "https://nosta.me/" -}} {{- end -}} {{- $anchor := printf `%s` $host $target $id ($label | htmlEscape) -}} {{/* Cloak first, so replacing the bare identifier cannot reach inside the anchor just inserted (the href and title both carry it). */}} {{- $cloak := printf "@@mention%d@@" $index -}} {{- $content = replace $content (printf "nostr:%s" $id) $cloak -}} {{- $content = replace $content $id $cloak -}} {{- $content = replace $content $cloak $anchor -}} {{- end -}} {{/* Put the undecodable mentions back as plain text — no link, since the identifier is malformed and would 404 — shortened to the same shape as a resolved fallback so a pasted-and-truncated id does not sprawl across the line. */}} {{- range $index, $id := $invalid -}} {{- $short := printf "%s…%s" (substr $id 0 10) (substr $id (sub (len $id) 4)) -}} {{- $content = replace $content (printf "@@nostr%d@@" $index) $short -}} {{- end -}} {{- $content | safeHTML -}}
